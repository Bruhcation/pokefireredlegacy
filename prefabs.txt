Legendary:
(
Map_Area_MapScripts::
	map_script MAP_SCRIPT_ON_TRANSITION, Map_Area_OnTransition
	map_script MAP_SCRIPT_ON_RESUME, Map_Area_OnResume
	.byte 0

Map_Area_OnResume::
	call_if_set FLAG_SYS_SPECIAL_WILD_BATTLE, Map_Area_EventScript_TryRemoveLegendary
	end

Map_Area_EventScript_TryRemoveLegendary::
	specialvar VAR_RESULT, GetBattleOutcome
	goto_if_ne VAR_RESULT, B_OUTCOME_CAUGHT, EventScript_Return
	removeobject VAR_LAST_TALKED
	return

Map_Area_OnTransition::
	call_if_unset FLAG_HIDE_LEGENDARY, Map_Area_EventScript_ShowLegendary
	end

Map_Area_EventScript_ShowLegendary::
	clearflag FLAG_HIDE_LEGENDARY
	return

Map_Area_EventScript_Legendary::
	goto_if_questlog EventScript_ReleaseEnd
	special QuestLog_CutRecording
	lock
	faceplayer
	setwildbattle SPECIES_LEGENDARY, 50
	waitse
	playmoncry SPECIES_LEGENDARY, CRY_MODE_ENCOUNTER
	message Text_Gyaoo
	waitmessage
	waitmoncry
	delay 10
	playbgm MUS_ENCOUNTER_GYM_LEADER, 0
	waitbuttonpress
	setflag FLAG_SYS_SPECIAL_WILD_BATTLE
	special StartLegendaryBattle
	waitstate
	clearflag FLAG_SYS_SPECIAL_WILD_BATTLE
	specialvar VAR_RESULT, GetBattleOutcome
	goto_if_eq VAR_RESULT, B_OUTCOME_WON, Map_Area_EventScript_DefeatedLegendary
	goto_if_eq VAR_RESULT, B_OUTCOME_RAN, Map_Area_EventScript_RanFromLegendary
	goto_if_eq VAR_RESULT, B_OUTCOME_PLAYER_TELEPORTED, Map_Area_EventScript_RanFromLegendary
	setflag FLAG_FOUGHT_LEGENDARY
	release
	end

Map_Area_EventScript_DefeatedLegendary::
	setflag FLAG_FOUGHT_LEGENDARY 
	goto EventScript_RemoveStaticMon
	end

Map_Area_EventScript_RanFromLegendary::
	setvar VAR_0x8004, SPECIES_LEGENDARY
	goto EventScript_MonFlewAway
	end
)

Tutors:

(
Map_Area_EventScript_MoveTutor::
	lock
	faceplayer
	msgbox Text_MoveTeach, MSGBOX_YESNO
	goto_if_eq VAR_RESULT, NO, EventScript_MoveDeclined
	call EventScript_CanOnlyBeLearnedOnce
	goto_if_eq VAR_RESULT, NO, EventScript_MoveDeclined
	msgbox Text_MoveWhichMon
	setvar VAR_0x8005, MOVETUTOR_MOVE
	call EventScript_ChooseMoveTutorMon
	goto_if_eq VAR_RESULT, FALSE, EventScript_MoveDeclined
	setflag FLAG_TUTOR_MOVE
	goto EventScript_MoveTaught
	end

EventScript_MoveDeclined::
	msgbox Text_MoveDeclined
	release
	end

EventScript_MoveTaught::
	msgbox Text_MoveTaught
	release
	end
)

Trades:

(

Map_Area_EventScript_Trader::
	lock
	faceplayer
	setvar VAR_0x8008, INGAME_TRADE_POKEMON
	call EventScript_GetInGameTradeSpeciesInfo
	goto_if_set FLAG_DID_POKEMON_TRADE, Map_Area_EventScript_AlreadyTraded
	msgbox Trade_Text_DoYouHaveMonWouldYouTradeForMon, MSGBOX_YESNO
	goto_if_eq VAR_RESULT, NO, Map_Area_EventScript_DeclineTrade
	call EventScript_ChooseMonForInGameTrade
	goto_if_ge VAR_0x8004, PARTY_SIZE, Map_Area_EventScript_DeclineTrade
	call EventScript_GetInGameTradeSpecies
	goto_if_ne VAR_RESULT, VAR_0x8009, Map_Area_EventScript_NotRequestedMon
	call EventScript_DoInGameTrade
	msgbox Trade_Text_Thanks
	setflag FLAG_DID_POKEMON_TRADE
	release
	end

Map_Area_EventScript_DeclineTrade::
	msgbox Trade_Text_WellIfYouDontWantTo
	release
	end

Map_Area_EventScript_NotRequestedMon::
	bufferspeciesname STR_VAR_1, VAR_0x8009
	msgbox Trade_Text_ThisIsntMon
	release
	end

Map_Area_EventScript_AlreadyTraded::
	msgbox Trade_Text_HasTradedMonGrownStronger
	release
	end
)
